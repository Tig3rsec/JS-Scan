Public proxies share IPs that firewalls (Cloudflare etc.) often block. A private Cloudflare Worker is the reliable fix: create one at dash.cloudflare.com, paste this, deploy, then put https://NAME.workers.dev/?url={url}&ua={ua} in the first box. Keep the worker URL private — it fetches whatever it is asked to. 
export default {
  async fetch(req) {
    const q = new URL(req.url).searchParams;
    const t = q.get('url');
    if (!t || !/^https?:\/\//i.test(t)) return new Response('missing ?url=', { status: 400 });
    const ua = q.get('ua') || 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/140.0.0.0 Safari/537.36';
    const r = await fetch(t, { redirect: 'follow', headers: {
      'User-Agent': ua, 'Accept': '*/*', 'Accept-Language': 'en-US,en;q=0.9' } });
    return new Response(r.body, { status: r.status, headers: {
      'Content-Type': r.headers.get('content-type') || 'text/plain',
      'Access-Control-Allow-Origin': '*' } });
  }
}
